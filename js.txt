function cargar(options) {
  options=options||{};
  var pageAgenda = document.getElementById('page-agenda');
  if (pageAgenda && !options.silencioso) pageAgenda.innerHTML = '<div class="loading">Cargando...</div>';

  google.script.run
    .withSuccessHandler(function(data) {
      data = normalizarRespuestaApi_(data);
      G.usuario = data.usuario || null;
      G.usuarios = data.usuarios || [];
      G.errores = data.errores || [];

      if (G.usuario && G.usuario.autorizado === false) {
        renderNoAutorizado_(G.usuario, data.meta || {});
        return;
      }

      if (!G.usuario) {
        diagnosticarCargaSinUsuario_(data);
        return;
      }

      G.oralidad = data.oralidad || [];
      G.debate = data.debate || [];
      G.participantes = data.participantes || [];
      G.config = data.config || {};
      G.historial = data.historial || [];
      G.imputados = data.imputados || [];
      normalizarDatosCliente_();
      CHUNKS_CARGADOS = {};
      if(data.meta && data.meta.ms) console.log('OFIJU carga inicial: '+data.meta.ms+' ms');
      mostrarDiagnosticoCarga_(data);

      setFechaUI();
      aplicarPermisosUI();
      renderAgenda();
      badges();
      actualizarNotificacionesUI();
      iniciarActividadUsuarioUI_();
      cargarEventosRevisionNotificaciones_();
      if(!options.silencioso && !G.accesoConfirmado) mostrarPantallaAcceso_();

      var activa = document.querySelector('.page.active');
      if (activa) {
        var id = activa.id || '';
        if (id === 'page-oralidad') renderOralidad();
        else if (id === 'page-debate') renderDebate();
        else if (id === 'page-expedientes') {
          var expBusq=document.getElementById('exp-buscar');
          var expVal=expBusq?expBusq.value:'';
          renderExpedientes();
          if(expVal){
            document.getElementById('exp-buscar').value=expVal;
            buscarExpediente();
          }
        }
        else if (id === 'page-recepcion') renderRecepcion();
        else if (id === 'page-personal') renderPersonal();
        else if (id === 'page-stats') renderStats();
        else if (id === 'page-simulador') renderSimulador();
        else if (id === 'page-busqueda') renderBusqueda();
        else if (id === 'page-revision') renderRevision();
        else if (id === 'page-admin') renderAdmin();
      }

      cargarSeccionesCriticas_(false);
    })
    .withFailureHandler(function(e) {
      var msg = e && e.message ? e.message : JSON.stringify(e);
      if (pageAgenda) {
        pageAgenda.innerHTML = '<div class="error-box">Error cargando datos: ' + msg + '</div>';
      }
      aviso('Error cargando datos');
    })
    .apiGetAllDataJson({ includeHistorial: !!options.includeHistorial });
}

function normalizarRespuestaApi_(data){
  if(typeof data==='string'){
    try{
      return JSON.parse(data);
    }catch(e){
      return {
        usuario:null,
        meta:{
          version:'',
          errorFatal:'No se pudo interpretar la respuesta JSON del servidor: '+(e.message||e),
          sesion:{}
        }
      };
    }
  }
  return data || {};
}

function diagnosticarCargaSinUsuario_(data){
  google.script.run
    .withSuccessHandler(function(diag){
      var meta=(data&&data.meta)||{};
      if(diag){
        meta.version=meta.version||diag.version||'';
        meta.sesion=meta.sesion||diag.sesion||{};
        meta.errorFatal=meta.errorFatal||'apiGetAllData devolvio una respuesta sin usuario. diagnosticoOFIJU si respondio desde la webapp.';
      }
      renderNoAutorizado_({
        autorizado:false,
        msg:'apiGetAllData devolvio informacion incompleta. Revisar duplicados de apiGetAllData/getAllData o mezcla de despliegue.',
        email:(diag&&diag.usuario&&diag.usuario.email)||''
      }, meta);
    })
    .withFailureHandler(function(e){
      renderNoAutorizado_({
        autorizado:false,
        msg:'apiGetAllData devolvio informacion incompleta y diagnosticoOFIJU no pudo ejecutarse desde la webapp.',
        email:''
      }, {
        version:(data&&data.meta&&data.meta.version)||'',
        errorFatal:e&&e.message?e.message:JSON.stringify(e||{}),
        sesion:(data&&data.meta&&data.meta.sesion)||{}
      });
    })
    .diagnosticoOFIJU();
}

function puedeCliente(accion){
  var rol=rolActual_();
  if(!rol) return false;
  if(rol==='SUPERADMIN') return true;

  var permisos={
    ADMIN:['VER','OPERAR','APROBAR_AGENDADO','GESTIONAR_PARTICIPANTES','VER_USUARIOS','GESTIONAR_USUARIOS','GESTIONAR_CONFIG','VER_LOGS','GESTIONAR_LOGS','ELIMINAR'],
    GESTOR:['VER','OPERAR','GESTIONAR_PARTICIPANTES'],
    CONSULTA:['VER']
  };

  return (permisos[rol]||[]).indexOf(accion)>=0;
}

function toggleNotificaciones(ev){
  if(ev&&ev.stopPropagation) ev.stopPropagation();
  var panel=document.getElementById('notif-panel');
  if(!panel) return;
  if(panel.style.display==='none' || !panel.style.display){
    cancelarCierreNotificaciones_();
    panel.innerHTML=renderPanelNotificaciones_();
    panel.style.display='block';
    prepararCierreNotificaciones_();
  }else{
    cerrarNotificaciones();
  }
}

function cerrarNotificaciones(){
  cancelarCierreNotificaciones_();
  var panel=document.getElementById('notif-panel');
  if(panel) panel.style.display='none';
}

function prepararCierreNotificaciones_(){
  var panel=document.getElementById('notif-panel');
  if(!panel) return;
  var wrap=panel.parentNode;
  panel.onmouseenter=cancelarCierreNotificaciones_;
  panel.onmouseleave=function(){ programarCierreNotificaciones_(); };
  if(wrap){
    wrap.onmouseenter=cancelarCierreNotificaciones_;
    wrap.onmouseleave=function(){ programarCierreNotificaciones_(); };
  }
}

function programarCierreNotificaciones_(){
  cancelarCierreNotificaciones_();
  NOTIF_CIERRE_TIMER=setTimeout(function(){
    cerrarNotificaciones();
  },3000);
}

function cancelarCierreNotificaciones_(){
  if(NOTIF_CIERRE_TIMER){
    clearTimeout(NOTIF_CIERRE_TIMER);
    NOTIF_CIERRE_TIMER=null;
  }
}

function actualizarNotificacionesUI(){
  var badge=document.getElementById('notif-badge');
  if(!badge) return;
  var items=notificacionesUsuario_();
  if(items.length){
    badge.textContent=items.length>9?'9+':items.length;
    badge.style.display='';
  }else{
    badge.style.display='none';
  }
  var panel=document.getElementById('notif-panel');
  if(panel&&panel.style.display!=='none') panel.innerHTML=renderPanelNotificaciones_();
}

function renderPanelNotificaciones_(){
  var items=notificacionesUsuario_();
  var h='<div class="notif-head"><div><strong>Notificaciones</strong><span>Solo lo relevante para tu usuario</span></div><div class="notif-actions"><button class="btn btn-sm" onclick="limpiarNotificacionesUI(event)">Borrar</button><button class="icon-btn mini" onclick="cerrarNotificaciones()">x</button></div></div>';
  if(!items.length) return h+'<div class="notif-empty">Sin novedades por ahora.</div>';
  h+='<div class="notif-list">';
  items.forEach(function(n){
    h+='<div class="notif-item '+escHtml(n.clase||'')+'" onclick="abrirNotificacion(\''+escJs(n.tipo||'')+'\',\''+escJs(n.exp||'')+'\',\''+escJs(n.hoja||'')+'\',\''+escJs(n.ref||'')+'\',\''+escJs(n.id||'')+'\')">';
    h+='<div class="notif-dot"></div><div class="notif-body"><strong>'+escHtml(n.titulo)+'</strong><span>'+escHtml(n.detalle||'')+'</span></div>';
    h+='</div>';
  });
  h+='</div>';
  return h;
}

function abrirNotificacion(tipo,exp,hoja,ref,id){
  if(id) marcarNotificacionLeida_(id);
  actualizarNotificacionesUI();
  cerrarNotificaciones();
  if(tipo==='revision-admin'){
    navegar('revision',document.getElementById('nav-revision'));
    return;
  }
  if(tipo==='errores'){
    navegar('admin',document.getElementById('nav-admin'));
    setTimeout(function(){scrollAElemento_(ref?('error-row-'+ref):'admin-errores-card');},80);
    return;
  }
  if(exp) verExpediente(exp,hoja||'oralidad');
}

function limpiarNotificacionesUI(ev){
  if(ev&&ev.stopPropagation) ev.stopPropagation();
  var items=notificacionesUsuario_({incluirLeidas:true});
  if(!items.length){
    cerrarNotificaciones();
    return;
  }
  var leidas=leerNotificacionesLeidas_();
  items.forEach(function(n){ if(n.id) leidas[n.id]=true; });
  guardarNotificacionesLeidas_(leidas);
  actualizarNotificacionesUI();
  var panel=document.getElementById('notif-panel');
  if(panel&&panel.style.display!=='none') panel.innerHTML=renderPanelNotificaciones_();
}

function claveNotificacionesUsuario_(){

function leerNotificacionesLeidas_(){
  try{
    var raw=localStorage.getItem(claveNotificacionesUsuario_());
    return raw?JSON.parse(raw):{};
  }catch(e){return {};}
}

function guardarNotificacionesLeidas_(obj){
  try{localStorage.setItem(claveNotificacionesUsuario_(),JSON.stringify(obj||{}));}catch(e){}
}

function marcarNotificacionLeida_(id){
  if(!id) return;
  var leidas=leerNotificacionesLeidas_();
  leidas[id]=true;
  guardarNotificacionesLeidas_(leidas);
}

function idNotificacion_(n){
  return normalizarBusqueda([n.tipo,n.titulo,n.exp,n.hoja,n.ref,n.detalle].join('|')).slice(0,180);
}

function aplicarEstadoNotificaciones_(items, incluirLeidas){
  var leidas=leerNotificacionesLeidas_();
  items.forEach(function(n){ n.id=idNotificacion_(n); });
  if(incluirLeidas) return items;
  return items.filter(function(n){ return !leidas[n.id]; });
}

  var propias=todas.filter(audienciaPerteneceUsuario_);
  var items=[];

  var venc=propias.filter(function(a){
    var fl=pF(a['FECHA LIMITE']);
    return fl&&fl<hoy&&estadoCuentaParaPlazo(a['ESTADO']);
  });
  if(venc.length) items.push({tipo:'lista',titulo:'Plazos vencidos propios',detalle:venc.length+' expediente(s) requieren atencion',clase:'danger',exp:expedienteAud_(venc[0]),hoja:venc[0]._t==='d'?'debate':'oralidad'});

  var crit=propias.filter(function(a){
    var fl=pF(a['FECHA LIMITE']);
    if(!fl) return false;
    var d=difDias(hoy,fl);
    return d>=0&&d<=3&&estadoCuentaParaPlazo(a['ESTADO']);
  });
  if(crit.length) items.push({tipo:'lista',titulo:'Plazos criticos propios',detalle:crit.length+' vencen dentro de 3 dias',clase:'warning',exp:expedienteAud_(crit[0]),hoja:crit[0]._t==='d'?'debate':'oralidad'});

  var prox=propias.filter(function(a){
    var f=pF(fechaAud_(a)), est=normalizarBusqueda(a['ESTADO']);
    return f&&f>=hoy&&f<=hasta&&est!==normalizarBusqueda('CANCELADA');
  }).sort(function(a,b){return (pF(fechaAud_(a))||0)-(pF(fechaAud_(b))||0);});
  if(prox.length) items.push({tipo:'audiencia',titulo:'Audiencias proximas',detalle:prox.length+' en los proximos 3 dias',clase:'ok',exp:expedienteAud_(prox[0]),hoja:prox[0]._t==='d'?'debate':'oralidad'});

  var revPropias=propias.filter(function(a){return normalizarBusqueda(a['ESTADO'])===normalizarBusqueda('EN REVISION');});
  if(revPropias.length) items.push({tipo:'audiencia',titulo:'Solicitudes en revision',detalle:revPropias.length+' esperando visto bueno',clase:'review',exp:expedienteAud_(revPropias[0]),hoja:revPropias[0]._t==='d'?'debate':'oralidad'});

  if(puedeCliente('APROBAR_AGENDADO')){
    var rev=pendientesRevision_();
    if(rev.length) items.push({tipo:'revision-admin',titulo:'Bandeja de revision',detalle:rev.length+' solicitud(es) para revisar',clase:'review'});
  }

  if(puedeCliente('VER_LOGS') && (G.errores||[]).length){
    var ult=(G.errores||[])[0]||{};
    items.push({tipo:'errores',titulo:'Errores registrados',detalle:(ult['FUNCION']?ult['FUNCION']+': ':'')+abrev(ult['ERROR']||((G.errores||[]).length+' evento(s) en el log tecnico'),56),clase:'warning',ref:ult['ID']||''});
  }

  items=items.concat(notificacionesEventosRevision_(todas));
  return aplicarEstadoNotificaciones_(items, !!opts.incluirLeidas).slice(0,8);
}

function notificacionesEventosRevision_(todas){
  var out=[];
  var vistos={};
  (G.eventosRevision||[]).forEach(function(ev){
    var accion=normalizarBusqueda(ev.ACCION||ev.Accion||'');
    if(accion!==normalizarBusqueda('APROBAR_AGENDADO') && accion!==normalizarBusqueda('DEVOLVER_REVISION')) return;
    var exp=ev.EXPEDIENTE||ev.Expediente||'';
    if(!exp) return;
    var fechaEv=pF(ev.FECHA||ev.Fecha||'');
    if(fechaEv && (G.hoy-fechaEv)>7*86400000) return;
    var key=accion+'|'+normalizarBusqueda(exp);
    if(vistos[key]) return;
    var aud=(todas||[]).filter(function(a){return normalizarBusqueda(expedienteAud_(a))===normalizarBusqueda(exp);})[0];
    var relevante=puedeCliente('APROBAR_AGENDADO') || (aud && audienciaPerteneceUsuario_(aud));
    if(!relevante) return;
    vistos[key]=true;
    var devuelta=accion===normalizarBusqueda('DEVOLVER_REVISION');
    out.push({
      tipo:'audiencia',
      titulo:devuelta?'Revision devuelta':'Revision aprobada',
      detalle:exp+(ev.FECHA?' - '+ev.FECHA:''),
      clase:devuelta?'warning':'ok',
      exp:exp,
      hoja:(ev.HOJA||'').toLowerCase()==='debate'?'debate':'oralidad'
    });
  });
  return out;
}

function clavesUsuarioActual_(){
  var keys=[], u=G.usuario||{};
  function add(v){
    var n=normalizarBusqueda(v);
    if(n&&keys.indexOf(n)===-1) keys.push(n);
  }
  add(u.email); add(u.nombre); add(u.emailActivo); add(u.emailEfectivo);
  (G.usuarios||[]).forEach(function(x){
    if(normalizarBusqueda(x.EMAIL||x.email)===normalizarBusqueda(u.email||'')){
      add(x.NOMBRE||x.nombre); add(x.AKA||x['A.K.A.']); add(x.AREA||x.area);
    }
  });
  (G.participantes||[]).forEach(function(p){
    if(normalizarBusqueda(p['EMAIL'])===normalizarBusqueda(u.email||'')){
      add(p['NOMBRE']); add(p['A.K.A.']); add(p['AREA']);
    }
  });
  return keys.filter(function(k){return k.length>1;});
}

function audienciaPerteneceUsuario_(a){

  return campos.some(function(c){
    var txt=normalizarBusqueda(a[c]);
    if(!txt) return false;
    return keys.some(function(k){return txt===k || txt.indexOf(k)>=0 || k.indexOf(txt)>=0;});
  });
}

function valorAud_(a, claves){
  a=a||{};
  for(var i=0;i<claves.length;i++){
    if(a[claves[i]]!=='' && a[claves[i]]!==null && typeof a[claves[i]]!=='undefined') return a[claves[i]];
  }
  return '';
}

function valorPorClaveFlexible_(a, incluye, excluye){
  a=a||{};
  incluye=(incluye||[]).map(function(x){return normalizarBusqueda(x);});
  excluye=(excluye||[]).map(function(x){return normalizarBusqueda(x);});
  for(var k in a){
    if(!Object.prototype.hasOwnProperty.call(a,k)) continue;
    var n=normalizarBusqueda(k);
    var ok=incluye.every(function(x){return n.indexOf(x)>=0;});
    var no=excluye.some(function(x){return n.indexOf(x)>=0;});
    if(ok && !no && a[k]!=='' && a[k]!==null && typeof a[k]!=='undefined') return a[k];
  }
  return '';
}

function getAgendaBucket_(tipo){

  if(tipo==='semana') return {titulo:'Audiencias de la semana visible', items:todas.filter(function(a){var f=pF(fechaAud_(a));return f&&f>=lun&&f<=vie;}), clase:'audiencias'};
  if(tipo==='vencidos') return {titulo:'Plazos vencidos', items:todas.filter(function(a){var fl=pF(a['FECHA LIMITE']);return fl&&fl<hoy&&estadoCuentaParaPlazo(a['ESTADO']);}), clase:'audiencias'};

  if(tipo==='no-disponibles'){
    var hoy0=new Date(hoy); hoy0.setHours(0,0,0,0);
    return {
      titulo:'Personas no disponibles hoy',
      items:G.participantes.filter(function(p){var ini=pF(p['INICIO AUSENCIA']),fin=pF(p['FIN AUSENCIA']);if(!p['NOMBRE']||!ini||!fin)return false;ini.setHours(0,0,0,0);fin.setHours(0,0,0,0);return hoy0>=ini&&hoy0<=fin;}),

    };
  }
  return {titulo:'Listado',items:[],clase:'audiencias'};
}

function inputFechaDate_(d){
  d=d instanceof Date?d:pF(d);
  if(!d) return '';
  var yyyy=d.getFullYear(), mm=d.getMonth()+1, dd=d.getDate();
  return yyyy+'-'+(mm<10?'0':'')+mm+'-'+(dd<10?'0':'')+dd;
}

function parseInputDateLocal_(valor){
  var m=String(valor||'').match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if(!m) return null;
  return new Date(parseInt(m[1],10),parseInt(m[2],10)-1,parseInt(m[3],10));
}

function rotuloFechaAgenda_(d){
  var ds=['Domingo','Lunes','Martes','Miercoles','Jueves','Viernes','Sabado'];
  return ds[d.getDay()]+' '+d.getDate()+'/'+(d.getMonth()+1)+'/'+d.getFullYear();
}

function salasConfiguradas_(){
  var base=(G.config.salas&&G.config.salas.length?G.config.salas:['SALA 1','SALA 2','SALA 3']).slice().filter(esValorSalaValido_);
  ['SALA 1','SALA 2','SALA 3','SALA ESPECIAL'].forEach(function(s){
    var existe=base.some(function(x){return normalizarBusqueda(x)===normalizarBusqueda(s);});
    if(!existe) base.push(s);
  });
  return base.filter(function(s,i,arr){
    return s && arr.findIndex(function(x){return normalizarBusqueda(x)===normalizarBusqueda(s);})===i;
  });
}

function esValorSalaValido_(sala){
  var s=normalizarBusqueda(sala);
  if(!s || /^\d+$/.test(s)) return false;
  return s.indexOf('sala')>=0 || s.indexOf('despacho')>=0 || s.indexOf('desp')>=0;
}

function esDespachoSala_(sala){
  var s=normalizarBusqueda(sala);
  return s.indexOf('despacho')>=0 || s.indexOf('desp')>=0;
}

function esSalaPrincipal_(sala){
  var s=normalizarBusqueda(sala);
  return s==='sala 1' || s==='sala 2' || s==='sala 3' || s==='sala especial';
}

function ordenarSalasAgenda_(salas){
  var orden={'sala 1':1,'sala 2':2,'sala 3':3,'sala especial':4};
  return salas.slice().sort(function(a,b){
    var oa=orden[normalizarBusqueda(a)]||50, ob=orden[normalizarBusqueda(b)]||50;
    if(oa!==ob) return oa-ob;
    return String(a).localeCompare(String(b));
  });
}

function htmlVentanasLibresSala_(auds){
  var inicio=8*60;
  var fin=14*60;
  var bloques=(auds||[]).map(function(a){
    var ini=parseMin(horaAud_(a));
    var dur=parseInt(a['DURACION']||60,10);
    if(!dur||isNaN(dur)) dur=60;
    return {ini:ini,fin:ini+dur};
  }).filter(function(b){return b.ini>0;}).sort(function(a,b){return a.ini-b.ini;});
  var cursor=inicio;
  var libres=[];
  bloques.forEach(function(b){
    if(b.ini>cursor) libres.push({ini:cursor,fin:b.ini});
    if(b.fin>cursor) cursor=b.fin;
  });
  if(cursor<fin) libres.push({ini:cursor,fin:fin});
  libres=libres.filter(function(b){return b.fin-b.ini>=15;});
  if(!libres.length) return '<div class="sala-free-empty">Sin ventanas libres entre 08:00 y 14:00.</div>';
  return '<div class="sala-free-list">'+libres.map(function(b){return '<span>'+minAHora(b.ini)+' - '+minAHora(b.fin)+'</span>';}).join('')+'</div>';
}

function htmlConsultaRecepcionRapida_(valor){

function buscarRecepcionRapidaItems_(valor){
  var q=normalizarBusqueda(valor||'');
  if(!q) return [];
  var grupos={};

  todasLasAuds().forEach(function(a){
    var exp=expedienteDeAudiencia_(a);
    if(!exp) return;
    var imp=a['IMPUTADO(S)']||a['IMPUTADO']||'';
    var dni=dniImputadoAudiencia_(a);
    var actoresValores=[
      a['DEFENSA'], a['FISCAL'], a['ASESOR - QTE'], a['ASESOR / QTE'],
      a['JUEZ'], a['AUDIENCISTA'], a['SECRETARIO'], a['GESTOR']
    ];
    var actores=actoresValores.map(function(v){return alternativasActorBusquedaCliente_(v).join(' ');}).join(' ');
    var blob=normalizarBusqueda(exp+' '+imp+' '+dni+' '+actores);
    if(blob.indexOf(q)<0) return;
    var k=normalizarBusqueda(exp);
    if(!grupos[k]) grupos[k]={exp:exp,imp:imp,dni:dni,items:[]};
    grupos[k].items.push(a);
    if(imp && imp.length>(grupos[k].imp||'').length) grupos[k].imp=imp;
    if(dni && !grupos[k].dni) grupos[k].dni=dni;
  });

  return Object.keys(grupos).map(function(k){
    var g=grupos[k];
    g.proxima=proximaAudienciaConsulta_(g.items);
    return g;
  }).sort(function(a,b){
    var ma=a.proxima&&a.proxima._ordenConsulta || 9999999999999;
    var mb=b.proxima&&b.proxima._ordenConsulta || 9999999999999;
    return ma-mb || String(a.exp).localeCompare(String(b.exp));
  });
}

function proximaAudienciaConsulta_(items){
  var ahora=new Date();
  var futuras=[];
  var pasadas=[];
  (items||[]).forEach(function(a){
    if(normalizarBusqueda(a['ESTADO'])===normalizarBusqueda('CANCELADA')) return;
    var f=pF(fechaAud_(a));
    if(!f) return;
    var min=parseMin(horaAud_(a)||'00:00');
    var dt=new Date(f.getFullYear(),f.getMonth(),f.getDate(),Math.floor(min/60),min%60);
    var c=copiar(a);
    c._ordenConsulta=dt.getTime();
    if(dt>=ahora) futuras.push(c); else pasadas.push(c);
  });
  futuras.sort(function(a,b){return a._ordenConsulta-b._ordenConsulta;});
  pasadas.sort(function(a,b){return b._ordenConsulta-a._ordenConsulta;});
  return futuras[0] || pasadas[0] || null;
}

function listaRecepcionFiltrada_(grupos){
  var qEl=document.getElementById('rec-buscar');
  var q=normalizarBusqueda(qEl?qEl.value:'');
  var tipo=G.recepcionTipo||'todos';
  return (grupos||[]).filter(function(g){
    if(g.ingreso) return false;
    if(q && normalizarBusqueda(g.exp+' '+g.imp).indexOf(q)<0) return false;
    if(tipo==='oralidad' && !g.tieneOralidad) return false;
    if(tipo==='debate' && !g.tieneDebate) return false;
    if(G.recepcionVencidos && !g.vencido) return false;
    if(G.recepcionPresos && !g.preso) return false;
    return true;
  }).sort(function(a,b){
    if(a.vencido!==b.vencido) return a.vencido?-1:1;
    if(a.critico!==b.critico) return a.critico?-1:1;
    return String(a.exp).localeCompare(String(b.exp));
  });
}

function recepcionGrupos_(){
  var grupos={};
  todasLasAuds().forEach(function(a){
    var exp=expedienteDeAudiencia_(a);
    if(!exp) return;
    var key=normalizarBusqueda(exp);
    if(!grupos[key]){
      grupos[key]={exp:exp,imp:'',items:[],ingreso:'',ingresoHora:'',ingresoKey:'',ingresoMs:0,tieneOralidad:false,tieneDebate:false,preso:false,vencido:false,critico:false,estado:'',tipoAud:'',tipoGrupo:'-',item:null,_ms:-1};
    }
    var g=grupos[key], imp=a['IMPUTADO(S)']||a['IMPUTADO']||'';
    g.items.push(a);
    if(imp && (!g.imp || imp.length>g.imp.length)) g.imp=imp;
    if(a._t==='d') g.tieneDebate=true; else g.tieneOralidad=true;
    if(a['PRESO']===true||a['PRESO']==='TRUE'||a['PRESO']==='true') g.preso=true;
    var fl=pF(a['FECHA LIMITE']);
    if(fl && estadoCuentaParaPlazo(a['ESTADO'])){
      var dias=difDias(G.hoy,fl);
      if(dias<0) g.vencido=true;
      else if(dias<=3) g.critico=true;
    }
    var ingreso=ingresoFisicoAudiencia_(a);
    if(ingreso){
      var fi=parseInputDateLocal_(ingreso)||pF(ingreso);
      var hi=horaIngresoFisicoAudiencia_(a);
      var ms=fi?fi.getTime():0;
      if(ms && hi){
        var hm=hi.split(':');
        ms += ((parseInt(hm[0],10)||0)*60 + (parseInt(hm[1],10)||0))*60000;
      }
      if(!g.ingreso || ms>=g.ingresoMs){
        g.ingreso=ingreso;
        g.ingresoHora=hi;
        g.ingresoKey=fi?inputFechaDate_(fi):'';
        g.ingresoMs=ms;
      }
    }
    var fecha=pF(fechaAud_(a))||pF(a['FECHA LIMITE']);
    var ms2=fecha?fecha.getTime():0;
    if(ms2>=g._ms){
      g._ms=ms2;
      g.item=a;
      g.estado=a['ESTADO']||'';
      g.tipoAud=tipoAud_(a)||'-';
    }
  });
  return Object.keys(grupos).map(function(k){
    var g=grupos[k];
    g.tipoGrupo=g.tieneOralidad&&g.tieneDebate?'Oralidad + Debate':(g.tieneDebate?'Debate':'Oralidad');
    return g;
  });
}

function buscarExpediente(){
  var input=document.getElementById('exp-buscar');
  var busq=(input?input.value:'').trim();

  if(!busq){
    document.getElementById('exp-resultado').innerHTML='';
    return;
  }

  var q=normalizarBusqueda(busq);
  var todas=[];

  G.oralidad.forEach(function(a){
    var c=copiar(a);
    c._t='oralidad';
    c._exp=c['NÂº EXPEDIENTE']||c['NÂº EXPTE']||'';
    c._imp=c['IMPUTADO(S)']||c['IMPUTADO']||'';

    if(normalizarBusqueda(c._exp+' '+c._imp).indexOf(q)>-1){
      todas.push(c);
    }
  });

  G.debate.forEach(function(a){
    var c=copiar(a);
    c._t='debate';
    c._exp=c['NÂº EXPTE']||c['NÂº EXPEDIENTE']||'';
    c._imp=c['IMPUTADO(S)']||c['IMPUTADO']||'';

    if(normalizarBusqueda(c._exp+' '+c._imp).indexOf(q)>-1){
      todas.push(c);
    }
  });

  if(!todas.length){
    document.getElementById('exp-resultado').innerHTML='<div class="alert-a">No se encontraron resultados para "'+escHtml(busq)+'"</div>';
    return;
  }

  var grupos={};

  todas.forEach(function(a){
    var key=normalizarBusqueda(a._exp);
    if(!key) return;

    if(!grupos[key]){
      grupos[key]={
        exp:a._exp,
        imp:a._imp,
        items:[]
      };
    }

    grupos[key].items.push(a);
  });

  var keys=Object.keys(grupos);

  if(keys.length===1){
    renderFichaExpediente(grupos[keys[0]].items);
  } else {
    renderListaCoincidenciasExpediente(grupos, busq);
  }
}

function formatearDetalleHistorial_(detalle){
  if(!detalle) return '';
  var s=String(detalle);
  try{
    var obj=JSON.parse(s);
    var partes=[];
    Object.keys(obj).forEach(function(k){
      if(obj[k]!=='' && obj[k]!==null && typeof obj[k]!=='undefined') partes.push(k+': '+obj[k]);
    });
    return partes.join(' | ') || s;
  }catch(e){
    return s;
  }
}

function calcularResumenExpediente(todas){
  var hoy=new Date();
  hoy.setHours(0,0,0,0);

  var realizadas=0, pendientes=0, canceladas=0;
  var ultimoEstado='';
  var prox=null;

  todas.forEach(function(a){
    var est=String(a['ESTADO']||'');
    if(est.indexOf('REALIZADA')>-1) realizadas++;
    else if(est.indexOf('CANCELADA')>-1) canceladas++;
    else pendientes++;

    if(est) ultimoEstado=est;

    var f=pF(a['DIA']||a['FECHA AUDIENCIA']);
    if(f){
      var ff=new Date(f); ff.setHours(0,0,0,0);
      if(ff>=hoy && (!prox || ff<prox)) prox=ff;
    }
  });

  return {
    realizadas:realizadas,
    pendientes:pendientes,
    canceladas:canceladas,
    ultimoEstado:ultimoEstado,
    proximaFecha:prox?fmt(prox):''
  };
}

function calcularAlertasExpediente(todas){
  var hoy=new Date();
  hoy.setHours(0,0,0,0);
  var alertas=[];

  var sinFecha=todas.filter(function(a){
    return !pF(a['DIA']||a['FECHA AUDIENCIA']) && estadoCuentaParaPlazo(a['ESTADO']);
  }).length;

  if(sinFecha>0) alertas.push(sinFecha+' intervencion(es) pendiente(s) sin fecha asignada.');

  var vencidas=todas.filter(function(a){
    var fl=pF(a['FECHA LIMITE']);
    return fl && fl<hoy && estadoCuentaParaPlazo(a['ESTADO']);
  }).length;

  if(vencidas>0) alertas.push(vencidas+' plazo(s) vencido(s).');

  var criticas=todas.filter(function(a){
    var fl=pF(a['FECHA LIMITE']);
    if(!fl || !estadoCuentaParaPlazo(a['ESTADO'])) return false;
    var d=difDias(hoy,fl);
    return d>=0 && d<=3;
  }).length;

  if(criticas>0) alertas.push(criticas+' plazo(s) criticos dentro de 3 dias.');

  var canceladas=todas.filter(function(a){ return String(a['ESTADO']||'').indexOf('CANCELADA')>-1; }).length;
  if(canceladas>0) alertas.push(canceladas+' intervencion(es) cancelada(s).');

  return alertas;
}

function calcularVinculosImputados_(todasExpediente, expteActual){
  var base={};
  imputadosExtraExpediente_(expteActual).forEach(function(p){
    var dni=normalizarDni_(p['DNI']);
    if(dni && !base['dni:'+dni]) base['dni:'+dni]={tipo:'dni',key:dni,nombre:'DNI '+p['DNI']};
    var nom=normalizarImputadoClave_(p['NOMBRE']);
    if(nom && !base['nom:'+nom]) base['nom:'+nom]={tipo:'nombre',key:nom,nombre:p['NOMBRE']};
  });

  (todasExpediente||[]).forEach(function(a){
    dnisDeAudiencia_(a).forEach(function(dni){
      var key=normalizarDni_(dni);
      if(key && !base['dni:'+key]) base['dni:'+key]={tipo:'dni',key:key,nombre:'DNI '+dni};
    });

    imputadosDeAudiencia_(a).forEach(function(nombre){
      var key=normalizarImputadoClave_(nombre);
      if(key && !base['nom:'+key]) base['nom:'+key]={tipo:'nombre',key:key,nombre:nombre};
    });
  });

  var keys=Object.keys(base);
  if(!keys.length) return [];

  var actualNorm=normalizarBusqueda(expteActual||'');
  var out=keys.map(function(k){return {tipo:base[k].tipo,key:base[k].key,nombre:base[k].nombre,expedientes:{}};});
  var todas=todasLasAuds();

  todas.forEach(function(a){
    var exp=expedienteDeAudiencia_(a);
    if(!exp || normalizarBusqueda(exp)===actualNorm) return;

    var imputados=imputadosDeAudiencia_(a);
    var dnis=dnisDeAudiencia_(a);
    imputadosExtraExpediente_(exp).forEach(function(p){
      if(p['NOMBRE']) imputados.push(p['NOMBRE']);
      if(p['DNI']) dnis.push(p['DNI']);
    });
    if(!imputados.length && !dnis.length) return;

    out.forEach(function(v){
      var coincide=v.tipo==='dni'
        ? dnis.some(function(dni){return normalizarDni_(dni)===v.key;})
        : imputados.some(function(nombre){return coincideImputado_(v.key, normalizarImputadoClave_(nombre));});
      if(!coincide) return;

      var expKey=normalizarBusqueda(exp);
      if(!v.expedientes[expKey]){
        v.expedientes[expKey]={
          exp:exp,
          imputadoRegistro:imputados.join(' / '),
          tramites:{},
          estados:{},
          total:0,
          ultima:null
        };
      }

      var item=v.expedientes[expKey];
      item.total++;
      item.tramites[a._t==='d'?'Debate':'Oralidad']=true;
      if(a['ESTADO']) item.estados[a['ESTADO']]=true;
      var fecha=pF(a['DIA']||a['FECHA AUDIENCIA']||a['FECHA LIMITE']);
      if(fecha && (!item.ultima || fecha>item.ultima)) item.ultima=fecha;
    });
  });

  out.forEach(function(v){
    v.lista=Object.keys(v.expedientes).map(function(k){return v.expedientes[k];})
      .sort(function(a,b){
        var fa=a.ultima||new Date(0), fb=b.ultima||new Date(0);
        return fb-fa || String(a.exp).localeCompare(String(b.exp));
      });
  });

  return out;
}

function separarImputados_(txt){
  var s=String(txt||'').replace(/\r/g,'\n').trim();
  if(!s) return [];

  var partes=s.split(/\n|;|\||\s+\/\s+|\s+-\s+|\s+y\s+/i);
  var vistos={}, out=[];

  partes.forEach(function(p){
    p=String(p||'').replace(/\s+/g,' ').trim();
    if(!p) return;
    var key=normalizarImputadoClave_(p);
    if(!key || vistos[key]) return;
    vistos[key]=true;
    out.push(p);
  });

  return out;
}

function normalizarImputadoClave_(nombre){
  return normalizarBusqueda(nombre)
    .replace(/[^\w\s]/g,' ')
    .replace(/\s+/g,' ')
    .trim();
}

function normalizarDni_(dni){
  return String(dni||'').replace(/\D/g,'');
}

function coincideImputado_(a,b){
  if(!a || !b) return false;
  if(a===b) return true;
  if(a.length>=8 && b.length>=8 && (a.indexOf(b)>=0 || b.indexOf(a)>=0)) return true;
  return false;
}

function resumirParticipantesExpediente(todas){
  var roles=[
    {k:'JUEZ',t:'Jueces'},
    {k:'FISCAL',t:'Fiscales'},
    {k:'DEFENSA',t:'Defensas'},
    {k:'AUDIENCISTA',t:'Audiencistas'},
    {k:'ASESOR - QTE',alt:'ASESOR / QTE',t:'Asesores / Querellantes'},
    {k:'SECRETARIO',t:'Secretarios'},
    {k:'GESTOR',t:'Gestores'}
  ];

  var out={};

  roles.forEach(function(r){
    out[r.t]={};
  });

  todas.forEach(function(a){
    roles.forEach(function(r){
      var v=a[r.k] || (r.alt?a[r.alt]:'');
      if(!v) return;

      var key=normalizarBusqueda(v);
      var contacto=buscarDatosContactoParticipante(v);

      if(!out[r.t][key]){
        out[r.t][key]={
          nombre:v,
          cantidad:0,
          telefono:contacto.telefono || '',
          email:contacto.email || '',
          recibe:contacto.recibe || '',
          obs:contacto.obs || ''
        };
      }

      out[r.t][key].cantidad++;

      // Si el primer match no tenia contacto pero otro nombre/alias si, completar.
      if(contacto.telefono && !out[r.t][key].telefono) out[r.t][key].telefono=contacto.telefono;

      if(contacto.obs && !out[r.t][key].obs) out[r.t][key].obs=contacto.obs;
    });
  });

  return out;
}

function buildActorPicker(pfx, fechaStr) {
  var roles=['JUEZ','FISCAL','DEFENSA','ABOGADO LITIGANTE','AUDIENCISTA','ASESOR','QUERELLANTE','SECRETARIO'];
  var h='';
  roles.forEach(function(rol){
    var ps=G.participantes.filter(function(p){return p['NOMBRE']&&p['ROL']===rol&&personaActiva(p);});
    if(!ps.length) return;
    var disponibles=ps.filter(function(p){return !participanteBloqueadoSim_(p, fechaStr);}).length;
    h+='<div class="actor-group role-'+rolClaseSim_(rol)+'"><div class="actor-group-title"><span class="actor-role-mark"></span><div><strong>'+escHtml(rolTituloSim_(rol))+'</strong><em>'+disponibles+' disponibles</em></div></div>';
    ps.forEach(function(p){
      var v=p['NOMBRE'];
      var nombreBase=limpiarTextoUI_(p['NOMBRE'])||labelParticipanteUI_(p);
      var meta=limpiarTextoUI_(p['AREA'])||limpiarTextoUI_(p['A.K.A.'])||rolTituloSim_(rol);

function fechaDesdeOffsetSim_(offset){
  var d=new Date((G&&G.hoy)?G.hoy:new Date());
  d.setHours(12,0,0,0);
  d.setDate(d.getDate()+(parseInt(offset,10)||0));

}

function fechaAInputSim_(d){
  var yyyy=d.getFullYear(), mm=d.getMonth()+1, dd=d.getDate();
  return yyyy+'-'+(mm<10?'0':'')+mm+'-'+(dd<10?'0':'')+dd;
}

function fechaComparableSim_(v){
  if(!v) return null;
  if(v instanceof Date && !isNaN(v.getTime())) return new Date(v.getFullYear(),v.getMonth(),v.getDate());
  var s=String(v).trim();
  var iso=s.match(/^(\d{4})-(\d{2})-(\d{2})$/);
  if(iso) return new Date(parseInt(iso[1],10),parseInt(iso[2],10)-1,parseInt(iso[3],10));
  var local=s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  if(local) return new Date(parseInt(local[3],10),parseInt(local[2],10)-1,parseInt(local[1],10));
  var d=pF(s);
  return d?new Date(d.getFullYear(),d.getMonth(),d.getDate()):null;
}

function participanteBloqueadoSim_(p, fechaStr){
  if(!p||!fechaStr) return null;
  var fecha=fechaComparableSim_(fechaStr);
  var inicio=fechaComparableSim_(p['INICIO AUSENCIA']);
  var fin=fechaComparableSim_(p['FIN AUSENCIA']);
  if(!fecha||!inicio||!fin) return null;
  if(fecha<inicio||fecha>fin) return null;
  var motivo=p['MOTIVO AUSENCIA']||'ausencia';
  return {
    motivo:motivo,
    detalle:'No disponible: '+motivo+' ('+fmt(inicio)+' al '+fmt(fin)+')'
  };
}

function escId(s){ return String(s||'').replace(/[^a-zA-Z0-9]/g,'_'); }

function toggleActor(pfx, val, nombre, rol, el){
  if(el&&el.getAttribute('data-disabled')==='1') return;
  var idx=-1;
  for(var i=0;i<ACTOR_SEL[pfx].length;i++){if(ACTOR_SEL[pfx][i].val===val){idx=i;break;}}
  if(idx>-1){
    ACTOR_SEL[pfx].splice(idx,1);
    el.classList.remove('selected');
  } else {
    ACTOR_SEL[pfx].push({val:val,nombre:nombre,rol:rol});
    el.classList.add('selected');
  }
  renderChips(pfx);
}

function renderChips(pfx){
  var chips=document.getElementById(pfx+'-chips');
  var count=document.getElementById(pfx+'-count');
  var total=(ACTOR_SEL[pfx]||[]).length;
  if(count) count.textContent=total+' participante'+(total===1?' seleccionado':'s seleccionados');
  if(!chips) return;
  chips.innerHTML=(ACTOR_SEL[pfx]||[]).map(function(a){
    return '<div class="chip">'+escHtml(a.nombre)+'<span class="chip-x" onclick="removeChip(\''+pfx+'\',\''+escapeAttr(a.val)+'\')">x</span></div>';
  }).join('');
}

function removeChip(pfx,val){
  for(var i=0;i<ACTOR_SEL[pfx].length;i++){
    if(ACTOR_SEL[pfx][i].val===val){
      ACTOR_SEL[pfx].splice(i,1);
      var el=document.getElementById(pfx+'-ai-'+escId(val));
      if(el) el.classList.remove('selected');
      break;
    }
  }
  renderChips(pfx);
}

function correrBusquedaGlobal(){
  var txt = normalizarBusqueda(document.getElementById('bg-texto') ? document.getElementById('bg-texto').value : '');
  var tipo = document.getElementById('bg-tipo') ? document.getElementById('bg-tipo').value : '';
  var estado = document.getElementById('bg-estado') ? document.getElementById('bg-estado').value : '';
  var sala = document.getElementById('bg-sala') ? document.getElementById('bg-sala').value : '';
  var desde = document.getElementById('bg-desde') ? document.getElementById('bg-desde').value : '';
  var hasta = document.getElementById('bg-hasta') ? document.getElementById('bg-hasta').value : '';

  var data = todasLasAuds();

  var resultados = data.filter(function(a){
    if(tipo && a._t !== tipo) return false;

    if(estado && normalizarBusqueda(a['ESTADO']) !== normalizarBusqueda(estado)) return false;

    var salaA = a['SALA'] || a['LUGAR CONFIRM.'] || '';
    if(sala && normalizarBusqueda(salaA) !== normalizarBusqueda(sala)) return false;

    var fechaA = pF(a['DIA'] || a['FECHA AUDIENCIA']);
    if(desde){
      var fd = new Date(desde + 'T12:00:00');
      if(!fechaA || fechaA < fd) return false;
    }
    if(hasta){
      var fh = new Date(hasta + 'T23:59:59');
      if(!fechaA || fechaA > fh) return false;
    }

    if(!txt) return true;

    var blob = camposBusqueda(a).join(' ');
    return normalizarBusqueda(blob).indexOf(txt) >= 0;
  });

  renderResultadosBusquedaGlobal(resultados, txt);
}

function normalizarBusqueda(v){
  return String(v || '')
    .trim()
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g,'')
    .replace(/\bn\s*[º°]?\s*(\d+)/g,'n $1')
    .replace(/\s+/g,' ');
}

function limpiarTextoUI_(v){
  return String(v || '').replace(/\s+/g,' ').trim();
}

function rolParticipanteCompatible_(rolPersona, rolObjetivo){
  var p=normalizarBusqueda(rolPersona);
  var r=normalizarBusqueda(rolObjetivo);
  if(!p || !r) return false;

}

function escHtml(v){
  return String(v || '')
    .replace(/&/g,'&amp;')
    .replace(/</g,'&lt;')
    .replace(/>/g,'&gt;')
    .replace(/"/g,'&quot;')
    .replace(/'/g,'&#039;');
}

function escJs(v){
  return String(v || '')
    .replace(/\\/g,'\\\\')
    .replace(/'/g,"\\'");
}

function propuestaRevisionDesdeHistorial_(a, hist){
  var exp=normalizarBusqueda(expedienteAud_(a));
  var hoja=normalizarBusqueda(a._hoja||'');
  var candidatos=(hist||[]).filter(function(h){
    return normalizarBusqueda(h['ACCION']||h['Accion']||'')===normalizarBusqueda('SOLICITAR_AGENDADO') &&
      (!exp || normalizarBusqueda(h['EXPEDIENTE']||h['Expediente']||'')===exp) &&
      (!hoja || normalizarBusqueda(h['HOJA']||h['Hoja']||'')===hoja);
  });
  if(!candidatos.length) return null;

  candidatos.sort(function(x,y){return (pF(hFechaHistorial_(y))||new Date(0))-(pF(hFechaHistorial_(x))||new Date(0));});
  var h=candidatos[0];
  var detalle=parseDetalleHistorialJson_(h['DETALLE']||h['Detalle']||'');
  var cambios=[];

  if(detalle && Array.isArray(detalle.cambios)){
    cambios=detalle.cambios.map(function(c){return {campo:c.campo, antes:c.antes, despues:c.despues};});
  }else if(detalle && detalle.campos){
    Object.keys(detalle.campos).forEach(function(campo){
      cambios.push({
        campo:campo,
        antes:normalizarBusqueda(campo)===normalizarBusqueda('ESTADO') ? (detalle.estado_previo||'') : 'Sin dato previo registrado',
        despues:normalizarBusqueda(campo)===normalizarBusqueda('ESTADO') ? (detalle.estado_solicitado||'AGENDADO') : detalle.campos[campo]
      });
    });
  }

  return {
    fecha:hFechaHistorial_(h)||'',
    usuario:h['USUARIO']||h['Usuario']||'',
    estadoSolicitado:(detalle&&detalle.estado_solicitado)||'AGENDADO',
    cambios:ordenarCambiosRevision_(cambios)
  };
}

function parseDetalleHistorialJson_(detalle){
  if(!detalle) return null;
  try{return JSON.parse(String(detalle));}catch(e){return null;}
}

function ordenarCambiosRevision_(cambios){
  var orden=['ESTADO','DIA','FECHA AUDIENCIA','HORA INICIO','HORA DE INICIO','DURACION','SALA','LUGAR CONFIRM.','JUEZ','AUDIENCISTA','SECRETARIO','FISCAL','DEFENSA','ASESOR - QTE','ASESOR / QTE','RESULTADO','OBSERVACIONES'];
  return (cambios||[]).filter(function(c){return c&&c.campo;}).sort(function(a,b){
    var ia=orden.indexOf(a.campo), ib=orden.indexOf(b.campo);
    if(ia<0) ia=999; if(ib<0) ib=999;
    return ia-ib || String(a.campo).localeCompare(String(b.campo));
  });
}

function alertasRevision_(a){
  var out=[];
  var fecha=pF(fechaAud_(a)), hora=horaAud_(a), sala=salaAud_(a), exp=expedienteAud_(a);
  if(!fecha) out.push('La audiencia no tiene fecha asignada.');
  if(!hora) out.push('La audiencia no tiene hora de inicio.');
  if(!sala) out.push('La audiencia no tiene sala asignada.');
  if(!a['JUEZ']) out.push('No tiene juez asignado.');
  if(a._hoja==='Oralidad' && !a['AUDIENCISTA']) out.push('No tiene audiencista asignado.');
  if(a._hoja==='Debate' && !a['SECRETARIO']) out.push('No tiene secretario asignado.');
  if(!fecha||!hora) return out;

  var ini=parseMin(hora), dur=parseInt(a['DURACION']||60,10);
  if(!dur||isNaN(dur)) dur=60;
  var fin=ini+dur;
  var actores=[];
  ['JUEZ','FISCAL','DEFENSA','AUDIENCISTA','SECRETARIO','ASESOR - QTE','ASESOR / QTE'].forEach(function(k){
    alternativasActorBusquedaCliente_(a[k]).forEach(function(v){ actores.push(normalizarBusqueda(v)); });
  });
  todasLasAuds().forEach(function(b){
    var expB=expedienteAud_(b);
    if(!expB || normalizarBusqueda(expB)===normalizarBusqueda(exp)) return;
    if(normalizarBusqueda(b['ESTADO'])===normalizarBusqueda('CANCELADA')) return;
    var fb=pF(fechaAud_(b));
    if(!fb||!mismaFecha(fb,fecha)) return;
    var ib=parseMin(horaAud_(b)), db=parseInt(b['DURACION']||60,10);
    if(!db||isNaN(db)) db=60;
    var seCruza=ini<ib+db && ib<fin;
    if(!seCruza) return;
    if(sala && normalizarBusqueda(salaAud_(b))===normalizarBusqueda(sala)){
      out.push('Posible conflicto de sala con expediente '+expB+'.');
    }
    var actoresB=[];
    ['JUEZ','FISCAL','DEFENSA','AUDIENCISTA','SECRETARIO','ASESOR - QTE','ASESOR / QTE'].forEach(function(k){
      alternativasActorBusquedaCliente_(b[k]).forEach(function(v){ actoresB.push(normalizarBusqueda(v)); });
    });
    actores.forEach(function(act){
      if(actoresB.indexOf(act)>=0) out.push('Posible conflicto de participante con expediente '+expB+'.');
    });
  });
  var vistos={};
  return out.filter(function(x){var k=normalizarBusqueda(x);if(vistos[k]) return false;vistos[k]=true;return true;}).slice(0,8);
}

function historialRevision_(expte){
  var q=normalizarBusqueda(expte||'');
  return (G.historial||[]).filter(function(h){
    return normalizarBusqueda(h['EXPEDIENTE']||h['Expediente']||'')===q;
  }).sort(function(a,b){
    return (pF(hFechaHistorial_(b))||new Date(0))-(pF(hFechaHistorial_(a))||new Date(0));
  });
}

function filtrarPersonas(){
  var tbody=document.getElementById('personas-tbody');
  if(!tbody) return;
  var txt=normalizarBusqueda(document.getElementById('per-buscar')?document.getElementById('per-buscar').value:'');
  var rol=document.getElementById('per-rol')?document.getElementById('per-rol').value:'';
  var ver=document.getElementById('per-activo')?document.getElementById('per-activo').value:'activos';
  var hoy=G.hoy;

  var data=G.participantes.filter(function(p){
    if(!p['NOMBRE']) return false;
    var activo=personaActiva(p);
    if(ver==='activos'&&!activo) return false;
    if(ver==='inactivos'&&activo) return false;
    if(rol&&p['ROL']!==rol) return false;
    if(txt){
      var blob=[p['NOMBRE'],p['A.K.A.'],p['ROL'],p['AREA'],p['TELEFONO'],p['EMAIL'],p['OBS CONTACTO']].join(' ');
      if(normalizarBusqueda(blob).indexOf(txt)===-1) return false;
    }
    return true;
  });

  data.sort(function(a,b){
    return String(a['ROL']||'').localeCompare(String(b['ROL']||'')) ||
           String(a['NOMBRE']||'').localeCompare(String(b['NOMBRE']||''));
  });

  var resumen=document.getElementById('personas-resumen');
  if(resumen) resumen.textContent=data.length+' persona(s) mostrada(s), agrupadas por rol general';

  if(!data.length){
    tbody.innerHTML='<tr><td colspan="13" style="color:#888;padding:14px">Sin resultados.</td></tr>';
    return;
  }

  var h='';
  var grupoActual='';
  data.forEach(function(p){
    var grupo=(p['ROL']||'SIN ROL');
    if(grupo!==grupoActual){
      grupoActual=grupo;
      h+='<tr><td colspan="13" style="background:#f8f7f4;color:#555;font-size:11px;font-weight:700;text-transform:uppercase;letter-spacing:.04em">'+escHtml(grupo)+'</td></tr>';
    }

    var ini=pF(p['INICIO AUSENCIA']),fin=pF(p['FIN AUSENCIA']);
    var aus=ini&&fin&&hoy>=ini&&hoy<=fin;
    var tieneBloqueo=!!(p['INICIO AUSENCIA']||p['FIN AUSENCIA']||p['MOTIVO AUSENCIA']);
    var activo=personaActiva(p);
    var notifica=(p['RECIBE NOTIFICACIONES']===true||String(p['RECIBE NOTIFICACIONES']).toUpperCase()==='TRUE');
    var contacto=[];
    if(p['TELEFONO']) contacto.push('Tel: '+escHtml(p['TELEFONO']));
    if(p['EMAIL']) contacto.push('Email: '+escHtml(p['EMAIL']));
    if(notifica) contacto.push('Notifica: si');
    if(p['OBS CONTACTO']) contacto.push('Obs: '+escHtml(abrev(p['OBS CONTACTO'],38)));

    h+='<tr>';
    h+='<td style="font-weight:500">'+escHtml(p['NOMBRE'])+'</td>';
    h+='<td style="color:#888">'+escHtml(p['A.K.A.']||'-')+'</td>';
    h+='<td>'+escHtml(p['ROL']||'-')+'</td>';
    h+='<td>'+escHtml(p['AREA']||'-')+'</td>';
    h+='<td style="font-size:11px;color:#666">'+(contacto.length?contacto.join('<br>'):'-')+'</td>';
    h+='<td>'+(activo?'<span class="est e-ag">Activo</span>':'<span class="est e-ca">Inactivo</span>')+'</td>';
    h+='<td>'+(aus?'<span class="est e-ca">No disponible</span>':'<span class="est e-ag">Disponible</span>')+'</td>';
    h+='<td>'+escHtml(p['MOTIVO AUSENCIA']||'-')+'</td><td>'+escHtml(p['INICIO AUSENCIA']||'-')+'</td><td>'+escHtml(p['FIN AUSENCIA']||'-')+'</td>';
    h+='<td>'+htmlSorteoPersona_(p)+'</td>';
    var maxDia=p['Maximo de Audiencias x dia']||p['MAX AUDIENCIAS/DIA']||p['MAXIMO DE AUDIENCIAS X DIA']||'';
    h+='<td style="text-align:center">'+(maxDia!==''?escHtml(maxDia):'-')+'</td>';
    h+='<td style="white-space:nowrap">';
    if(puedeCliente('GESTIONAR_PARTICIPANTES')){
      h+='<button class="btn btn-sm" onclick="abrirEditarPersona(\''+escapeAttr(p['NOMBRE'])+'\')">Editar</button> ';
      h+='<button class="btn btn-sm" onclick="abrirBloqueo(\''+escapeAttr(p['NOMBRE'])+'\')">'+(tieneBloqueo?'Bloqueo':'Bloquear')+'</button>';
      if(tieneBloqueo) h+=' <button class="btn btn-sm btn-danger" onclick="borrarBloqueo(\''+escapeAttr(p['NOMBRE'])+'\')">Borrar</button>';

function htmlSorteoPersona_(p){
  var rol=p['ROL']||'';
  if(rol!=='JUEZ' && rol!=='AUDIENCISTA') return '<span style="color:#888">-</span>';
  var c=cargaSorteoPersona_(p);
  var alta=p['FECHA ALTA SORTEO']||'';
  var altaDate=fechaSorteoDateCliente_(alta);
  var altaTxt=altaDate ? fmt(altaDate) : 'Sin alta';
  var cls=c.alerta?' sorteo-warn':'';
  return '<div class="sorteo-cell'+cls+'"><strong>'+escHtml(altaTxt)+'</strong><span>Hoy '+c.hoy+' | 7d '+c.d7+' | 30d '+c.d30+'</span><em>Ventana 30d: '+c.dias30+' dias activos</em></div>';
}

function cargaSorteoPersona_(p){
  var rol=p['ROL']==='AUDIENCISTA'?'AUDIENCISTA':'JUEZ';
  var hoy=new Date(G.hoy); hoy.setHours(0,0,0,0);
  var desde7=new Date(hoy); desde7.setDate(hoy.getDate()-6);
  var desde30=new Date(hoy); desde30.setDate(hoy.getDate()-29);
  var out={hoy:0,d7:0,d30:0,dias30:diasActivosSorteoCliente_(p, desde30, hoy),alerta:false};

  (G.oralidad||[]).forEach(function(a){
    if(!estadoValidoSorteoCliente_(a['ESTADO'])) return;
    var f=fechaSorteoDateCliente_(fechaAud_(a));
    if(!f) return;
    f.setHours(0,0,0,0);
    if(f>hoy || f<desde30) return;
    var valor=rol==='AUDIENCISTA' ? (a['AUDIENCISTA']||'') : (a['JUEZ']||'');
    if(!actorCoincidePersonaSorteo_(valor, p, rol)) return;
    out.d30++;
    if(f>=desde7) out.d7++;
    if(f.getTime()===hoy.getTime()) out.hoy++;
  });

  out.alerta=out.dias30>0 && (out.d30/out.dias30)>0.7;
  return out;
}

function diasActivosSorteoCliente_(p, desde, hasta){
  var alta=fechaSorteoDateCliente_(p['FECHA ALTA SORTEO']||p['ACTIVO DESDE']||p['FECHA ALTA']||p['CREADO']||'');
  var inicio=alta && alta>desde ? alta : desde;
  inicio.setHours(0,0,0,0);
  hasta=new Date(hasta); hasta.setHours(0,0,0,0);
  return Math.max(1, Math.floor((hasta.getTime()-inicio.getTime())/86400000)+1);
}

function fechaSorteoDateCliente_(v){
  if(!v) return null;
  if(v instanceof Date) return isNaN(v.getTime()) ? null : new Date(v.getFullYear(), v.getMonth(), v.getDate());
  var s=String(v).trim();
  if(!s) return null;
  return parseInputDateLocal_(s) || pF(s);
}

function estadoValidoSorteoCliente_(estado){
  var e=normalizarBusqueda(estado);
  return e && e!==normalizarBusqueda('CANCELADA') && e!==normalizarBusqueda('SIN AGENDA');
}

function actorCoincidePersonaSorteo_(valor, p, rol){
  var partes=splitActoresCampo_(valor).map(normalizarBusqueda);
  if(!partes.length) partes=[normalizarBusqueda(valor)];
  var vals=[
    valorParticipantePorRol_(p, rol),
    p['NOMBRE'],
    p['A.K.A.'],
    p['AREA']
  ].map(normalizarBusqueda).filter(Boolean);
  return partes.some(function(x){return vals.indexOf(x)>=0;});
}

function personaActiva(p){
  if(!p) return false;
  var v=p['ACTIVO'];
  return v===true || String(v).toUpperCase()==='TRUE' || String(v).trim()==='1';
}

function optsRolesPersonas(){

  roles.sort();

}

function cargarStats(){
  var toda=todasLasAuds();

  var est=document.getElementById('st-est').value;
  var res=document.getElementById('st-res').value;
  var sala=document.getElementById('st-sala').value;
  var juez=document.getElementById('st-juez').value;
  var fiscal=document.getElementById('st-fiscal').value;
  var gestor=document.getElementById('st-gestor').value;
  var audiencista=document.getElementById('st-audiencista').value;
  var pr=document.getElementById('st-preso').value;
  var desde=document.getElementById('st-desde').value;
  var hasta=document.getElementById('st-hasta').value;
  var filtradas=toda.filter(function(a){
    if(tipo){var t=a['TIPO DE AUD.']||a['TIPO DE AUD']||'';if(t!==tipo)return false;}
    if(est&&a['ESTADO']!==est) return false;
    if(res&&a['RESULTADO']!==res) return false;
    if(sala){var s=a['SALA']||a['LUGAR CONFIRM.']||'';if(s!==sala)return false;}
    if(juez&&a['JUEZ']!==juez) return false;
    if(fiscal&&a['FISCAL']!==fiscal) return false;
    if(gestor&&a['GESTOR']!==gestor) return false;
    if(audiencista&&a['AUDIENCISTA']!==audiencista) return false;
    if(pr==='si'&&!(a['PRESO']===true||a['PRESO']==='TRUE')) return false;
    if(pr==='no'&&(a['PRESO']===true||a['PRESO']==='TRUE')) return false;
    if(desde||hasta){var f=pF(a['DIA']||a['FECHA AUDIENCIA']);if(desde&&f&&f<new Date(desde+'T12:00:00'))return false;if(hasta&&f&&f>new Date(hasta+'T23:59:59'))return false;}
    return true;
  });
  function contarPorValor(fn){
    var c={};
    filtradas.forEach(function(a){
      var v=fn(a)||'Sin datos';
      c[v]=(c[v]||0)+1;
    });
    return c;
  }
  var total=filtradas.length;
  var totalExistentes=toda.length;

  var porResultado=contarPorValor(function(a){return a['RESULTADO'];});
  var porTipo=contarPorValor(function(a){return tipoAud_(a);});

  var porAudiencista=contarPorValor(function(a){return a['AUDIENCISTA'];});
  var porPreso={'Con preso':conP,'Sin preso':total-conP};

  };
}

function statGroup(titulo,obj,total){
  var items=[]; for(var k in obj) if(k&&k!=='Sin datos'&&obj[k]>0) items.push({k:k,v:obj[k]});
  if(!items.length) return '';
  items.sort(function(a,b){return b.v-a.v;});
  var h='<div class="card"><div class="card-title">'+titulo+'</div>';
  items.slice(0,10).forEach(function(it){
    var pct=total>0?Math.round(it.v/total*100):0;
    h+='<div class="stat-row"><span>'+abrev(it.k,35)+'</span><strong>'+it.v+' <span style="font-weight:400;color:#aaa;font-size:10px">'+pct+'% filtro</span></strong></div>';
    h+='<div class="stat-bar"><div class="stat-bar-fill" style="width:'+pct+'%"></div></div>';
  });
  return h+'</div>';
}

function valorParticipantePorRol_(p, rol){
  if(!p) return '';
  if(rol === 'FISCAL') return p['AREA'] || '';
  if(rol === 'GESTOR' || rol === 'AUDIENCISTA') return p['A.K.A.'] || p['NOMBRE'] || '';
  return p['NOMBRE'] || '';
}

function labelParticipante_(p, rol){
  return labelParticipanteUI_(p);
}

function participanteSeleccionado_(p, rol, valAct){
  var buscado=normalizarBusqueda(valAct||'');
  if(!buscado) return false;
  var vals=[
    valorParticipantePorRol_(p, rol),
    p['NOMBRE'],
    p['A.K.A.'],
    p['AREA']
  ];
  return vals.some(function(v){return normalizarBusqueda(v||'')===buscado;});
}

function esDefensaParticularRegistrada_(valor){
  return normalizarBusqueda(valor).indexOf('particular - ')===0;
}

function valorDefensaLitigante_(p){
  var nombre=limpiarTextoUI_(p&&p['NOMBRE']);
  return nombre ? 'PARTICULAR - '+nombre : '';
}

function participanteSeleccionadoDefensa_(p, valAct){
  if(!p) return false;
  var buscado=normalizarBusqueda(valAct||'');
  if(!buscado) return false;
  var vals=[
    valorParticipantePorRol_(p, 'DEFENSA'),
    p['NOMBRE'],
    p['A.K.A.'],
    p['AREA']
  ];
  if(normalizarBusqueda(p['ROL'])===normalizarBusqueda('ABOGADO LITIGANTE')){
    vals.push(valorDefensaLitigante_(p));
  }
  return vals.some(function(v){return normalizarBusqueda(v||'')===buscado;});
}

function opcionesParticipantesPorRol_(rol, valAct, fechaStr){
  var r='<option value="">-- Sin asignar --</option>';
  var matched=!valAct;
  if(rol==='DEFENSA'){
    var particularSel=normalizarBusqueda(valAct)===normalizarBusqueda('PARTICULAR');
    if(particularSel) matched=true;
    r+='<option value="PARTICULAR"'+(particularSel?' selected':'')+'>PARTICULAR</option>';
  }
  G.participantes
    .filter(function(p){return p['NOMBRE'] && rolParticipanteCompatible_(p['ROL'], rol) && personaActiva(p);})
    .sort(function(a,b){
      return String(a['ROL']||'').localeCompare(String(b['ROL']||'')) ||
             String(a['AREA']||'').localeCompare(String(b['AREA']||'')) ||
             String(a['NOMBRE']||'').localeCompare(String(b['NOMBRE']||''));
    })
    .forEach(function(p){
      var esLitigante=rol==='DEFENSA' && normalizarBusqueda(p['ROL'])===normalizarBusqueda('ABOGADO LITIGANTE');
      var v=esLitigante ? valorDefensaLitigante_(p) : valorParticipantePorRol_(p, rol);
      var label=(esLitigante?'PARTICULAR - ':'')+labelParticipante_(p, rol);
      var sel=rol==='DEFENSA' ? participanteSeleccionadoDefensa_(p, valAct) : participanteSeleccionado_(p, rol, valAct);
      if(sel) matched=true;
      var bloqueo=participanteBloqueadoSim_(p, fechaStr);
      var disabled=bloqueo&&!sel;
      r+='<option value="'+escapeAttr(v)+'"'+(sel?' selected':'')+(disabled?' disabled':'')+'>'+escHtml(label+(bloqueo?' - '+bloqueo.detalle:''))+'</option>';
    });
  if(valAct && !matched){
    r+='<option value="'+escapeAttr(valAct)+'" selected>'+escHtml(valAct)+'</option>';
  }
  return r;
}

function opcionesAbogadosLitigantes_(valAct){
  var r='<option value="">-- Seleccionar abogado --</option>';
  var valNorm=normalizarBusqueda(valAct||'');
  G.participantes
    .filter(function(p){return p['NOMBRE'] && normalizarBusqueda(p['ROL'])===normalizarBusqueda('ABOGADO LITIGANTE') && personaActiva(p);})
    .sort(function(a,b){return String(a['NOMBRE']||'').localeCompare(String(b['NOMBRE']||''));})
    .forEach(function(p){
      var v=valorDefensaLitigante_(p);
      var sel=valNorm && (normalizarBusqueda(v)===valNorm || normalizarBusqueda(p['NOMBRE'])===valNorm);
      r+='<option value="'+escapeAttr(v)+'"'+(sel?' selected':'')+'>'+escHtml(labelParticipanteUI_(p))+'</option>';
    });
  return r;
}

function inicializarAsesoresEdicion_(valor){
  EDIT_ASESORES=[];
  splitActoresCampo_(valor).forEach(function(v){
    if(!v) return;
    var p=buscarParticipantePorTexto_(v);
    var val=p ? p['NOMBRE'] : v;
    var label=p ? labelParticipante_(p, p['ROL']) : v;
    agregarAsesorEdicion_(val, label, true);
  });
}

function buscarParticipantePorTexto_(txt){
  var q=normalizarBusqueda(limpiarPrefijoParticular_(txt));
  return (G.participantes||[]).filter(function(p){
    return normalizarBusqueda(p['NOMBRE'])===q || normalizarBusqueda(p['A.K.A.'])===q || normalizarBusqueda(p['AREA'])===q;
  })[0] || null;
}

function limpiarPrefijoParticular_(txt){
  return String(txt||'').replace(/^\s*particular\s*-\s*/i,'').trim();
}

function alternativasActorBusquedaCliente_(valor){
  var out=[];
  splitActoresCampo_(valor).forEach(function(v){
    if(v) out.push(v);
    var limpio=limpiarPrefijoParticular_(v);
    if(limpio && limpio!==v) out.push(limpio);
    var p=buscarParticipantePorTexto_(v);
    if(p){
      out.push(p['NOMBRE'], p['A.K.A.'], p['AREA'], labelParticipanteUI_(p));
    }
  });
  return out.filter(function(x,i,arr){
    return x && arr.indexOf(x)===i;
  });
}

}

function cargarChunkDatos_(tipo, forzar){
  if(!forzar && CHUNKS_CARGADOS[tipo]) return;
  CHUNKS_CARGADOS[tipo]=true;

  google.script.run
    .withSuccessHandler(function(resp){
      var data=normalizarRespuestaApi_(resp);
      if(!data || data.ok===false){
        mostrarAvisoDatosParciales_(tipo, (data&&data.msg)||'No se pudo cargar');
        return;
      }

      if(tipo==='debate') G.debate=data.items||[];
      else if(tipo==='participantes') G.participantes=data.items||[];

      normalizarDatosCliente_();
      console.log('OFIJU chunk '+tipo+': '+((data.items||[]).length)+' item(s)');
      if(!(data.items||[]).length) mostrarAvisoDatosParciales_(tipo, 'la hoja devolvio 0 registros');
      badges();
      renderTrasMutacion_();
      mostrarDiagnosticoDatosCliente_('chunk '+tipo);
    })
    .withFailureHandler(function(e){
      mostrarAvisoDatosParciales_(tipo, e&&e.message?e.message:JSON.stringify(e||{}));
    })
    .apiGetDataChunkJson(tipo);
}

function opcionesEstadoEdicion_(estadoActual){
  var estados=G.config.estados||['PENDIENTE - EN CURSO','EN REVISION','AGENDADO','REALIZADA','CANCELADA','SUSP. Y REPROGRAMADA','SIN AGENDA'];
  var actualNorm=normalizarBusqueda(estadoActual);
  var acciones=[normalizarBusqueda('AGENDADO'),normalizarBusqueda('CANCELADA')];
  var out=[];
  estados.forEach(function(e){
    var n=normalizarBusqueda(e);
    if(acciones.indexOf(n)>=0 && n!==actualNorm) return;
    if(out.indexOf(e)===-1) out.push(e);
  });
  if(estadoActual && out.map(normalizarBusqueda).indexOf(actualNorm)<0) out.unshift(estadoActual);
  return out.map(function(e){
    return '<option value="'+escapeAttr(e)+'"'+(normalizarBusqueda(e)===actualNorm?' selected':'')+'>'+escHtml(e)+'</option>';
  }).join('');
}

function guardarEdicion(estadoOverride){

  var isO=hoja==='Oralidad';
  var campos={};

  if(isO) campos['AUDIENCISTA']=document.getElementById('edit-audiencista').value;
  campos['FISCAL']=document.getElementById('edit-fiscal').value;
  campos['DEFENSA']=document.getElementById('edit-defensa').value;
  campos['ASESOR - QTE']=(EDIT_ASESORES||[]).map(function(x){return x.val;}).join('; ');

  }

  google.script.run
    .withSuccessHandler(function(r){

        refrescarDatosEnSegundoPlano_();

      }

    })
    .actualizarExpediente({
      hoja:hoja,

    });

function accionRapidaEditar(accion){
  if(accion==='cancelar'){
    confirmarOfiju({
      kicker:'Audiencia',
      titulo:'Cancelar intervencion',
      mensaje:'Cancelar esta intervencion?',
      confirmText:'Cancelar intervencion',
      danger:true,
      onConfirm:function(){ guardarEdicion('CANCELADA'); }
    });
    return;
  }

  if(accion==='agendar'){
    confirmarOfiju({
      kicker:'Revision',
      titulo:'Enviar a revision',
      mensaje:'Enviar esta intervencion a revision para agendar?',
      confirmText:'Enviar a revision',
      onConfirm:function(){ guardarEdicion('AGENDADO'); }
    });
  }
}

function accionRapidaDesdeFicha(expte,tipo,accion){
  var esDebate=tipo==='debate'||tipo==='d';
  var hoja=esDebate?'Debate':'Oralidad';
  var estado=accion==='cancelar'?'CANCELADA':'AGENDADO';
  var pregunta=accion==='cancelar'
    ? 'Cancelar esta intervencion?'
    : 'Enviar esta intervencion a revision para agendar?';

  confirmarOfiju({
    kicker:accion==='cancelar'?'Audiencia':'Revision',
    titulo:accion==='cancelar'?'Cancelar intervencion':'Enviar a revision',
    mensaje:pregunta,
    detalle:expte||'',
    confirmText:accion==='cancelar'?'Cancelar intervencion':'Enviar a revision',
    danger:accion==='cancelar',
    onConfirm:function(){
      google.script.run
        .withSuccessHandler(function(r){
          if(r&&r.ok){
            aplicarItemLocal_(hoja, r.item, expte, {'ESTADO':estado});
            renderTrasMutacion_();
            refrescarDatosEnSegundoPlano_();
            toast(r.enRevision ? 'Enviado a revision para agendar' : (accion==='cancelar'?'Intervencion cancelada':'Agendamiento actualizado'));
          }else{
            aviso((r&&r.msg)||'No se pudo realizar la accion');
          }
        })
        .withFailureHandler(function(e){
          aviso('Error: '+(e.message||JSON.stringify(e)));
        })
        .actualizarExpediente({
          hoja:hoja,
          nro_expediente:expte,
          campos:{ESTADO:estado}
        });
    }
  });
}

function aplicarItemLocal_(hoja, item, expte, fallback){
  var arr=hoja==='Debate'?G.debate:G.oralidad;
  var buscado=normalizarBusqueda(expte || expedienteDeAudiencia_(item||{}) || (fallback&&fallback.nro_expediente) || '');
  if(!buscado) return;

  var nuevo=item ? copiar(item) : {};
  if(!item && fallback){
    nuevo=audienciaLocalDesdeFallback_(hoja, fallback, expte);
  }

  nuevo._t=hoja==='Debate'?'d':'o';

  var idx=-1;
  for(var i=0;i<arr.length;i++){
    if(normalizarBusqueda(expedienteDeAudiencia_(arr[i]))===buscado){idx=i;break;}
  }

  if(idx>=0){
    arr[idx]=Object.assign({}, arr[idx], nuevo);
  }else{
    arr.unshift(nuevo);
  }
}
